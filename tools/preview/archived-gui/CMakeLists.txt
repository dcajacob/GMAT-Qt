# Build only the Qt desktop against the archived engine libraries.
cmake_minimum_required(VERSION 3.21)
project(GmatPackagedGuiFix LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(GMAT_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
file(GLOB_RECURSE gmat_headers "${GMAT_SOURCE_DIR}/src/base/*.hpp" "${GMAT_SOURCE_DIR}/src/gmatutil/*.hpp")
set(gmat_includes "${CSPICE_DIR}/include")
foreach(header IN LISTS gmat_headers)
  get_filename_component(directory "${header}" DIRECTORY)
  list(APPEND gmat_includes "${directory}")
endforeach()
list(REMOVE_DUPLICATES gmat_includes)
foreach(native GmatUtil GmatBase)
  add_library(${native} SHARED IMPORTED)
  set_target_properties(${native} PROPERTIES IMPORTED_LOCATION "${GMAT_LIBRARY_${native}}" INTERFACE_INCLUDE_DIRECTORIES "${gmat_includes}")
  if(WIN32)
    set_target_properties(${native} PROPERTIES IMPORTED_IMPLIB "${GMAT_IMPLIB_${native}}")
  endif()
endforeach()
add_compile_definitions(__USE_SPICE__ USE_64_BIT_LONGS UNICODE _UNICODE)
if(WIN32)
  add_compile_definitions(NOMINMAX)
  add_compile_options(/Zc:__cplusplus /MP3)
else()
  add_compile_definitions(LINUX_MAC NO_GCC_PRAGMA)
  add_compile_options(-fno-strict-aliasing)
endif()
set(GMAT_QT_TUTORIAL_ONLY ON)
add_subdirectory("${GMAT_SOURCE_DIR}/src/qtgui" frontend)
add_executable(GmatQt "${GMAT_SOURCE_DIR}/src/qtgui/main.cpp")
target_link_libraries(GmatQt PRIVATE GmatQtFrontend)
if(APPLE)
  set_target_properties(GmatQt PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path/../Frameworks;@executable_path")
elseif(UNIX)
  set_target_properties(GmatQt PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN;$ORIGIN/../lib")
endif()
