cmake_minimum_required(VERSION 3.21)
project(GmatPackagedTutorialProbe LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(GMAT_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(GMAT_INSTALLATION "" CACHE PATH "Extracted GMAT Qt release installation")
set(CSPICE_DIR "" CACHE PATH "CSPICE headers matching the release")
if(NOT IS_DIRECTORY "${GMAT_INSTALLATION}")
   message(FATAL_ERROR "GMAT_INSTALLATION must name an extracted release")
endif()

if(APPLE)
   set(gmat_binary_dir "${GMAT_INSTALLATION}/Contents/Frameworks")
else()
   set(gmat_binary_dir "${GMAT_INSTALLATION}/bin")
endif()

file(GLOB gmat_base_directories LIST_DIRECTORIES true "${GMAT_SOURCE_DIR}/src/base/*")
file(GLOB gmat_util_directories LIST_DIRECTORIES true "${GMAT_SOURCE_DIR}/src/gmatutil/*")
set(gmat_header_directories "${GMAT_SOURCE_DIR}/src/base/forcemodel/harmonic"
   "${GMAT_SOURCE_DIR}/src/base/factory/guicomponents"
   "${GMAT_SOURCE_DIR}/src/gmatutil/util/datawriter"
   "${GMAT_SOURCE_DIR}/src/gmatutil/util/interpolator"
   "${GMAT_SOURCE_DIR}/src/gmatutil/util/matrixoperations")
list(APPEND gmat_header_directories "${CSPICE_DIR}/include")
foreach(directory IN LISTS gmat_base_directories gmat_util_directories)
   if(IS_DIRECTORY "${directory}")
      list(APPEND gmat_header_directories "${directory}")
   endif()
endforeach()

option(GMAT_TUTORIAL_BUILD_BASE "Build the affected core against archived utility libraries" OFF)
foreach(name GmatUtil GmatBase)
   if(GMAT_TUTORIAL_BUILD_BASE AND (name STREQUAL "GmatBase" OR APPLE))
      continue()
   endif()
   if(WIN32)
      find_library(gmat_${name} NAMES ${name} lib${name} PATHS "${GMAT_IMPORT_LIBRARIES}" NO_DEFAULT_PATH REQUIRED)
   elseif(APPLE)
      find_library(gmat_${name} NAMES ${name}.R2026a PATHS "${gmat_binary_dir}" NO_DEFAULT_PATH REQUIRED)
   else()
      find_library(gmat_${name} NAMES ${name} PATHS "${gmat_binary_dir}" NO_DEFAULT_PATH REQUIRED)
   endif()
   add_library(${name} UNKNOWN IMPORTED)
   set_target_properties(${name} PROPERTIES
      IMPORTED_LOCATION "${gmat_${name}}"
      INTERFACE_INCLUDE_DIRECTORIES "${gmat_header_directories}")
endforeach()

add_compile_definitions(__USE_SPICE__ USE_64_BIT_LONGS UNICODE _UNICODE)
if(WIN32)
   add_compile_definitions(NOMINMAX)
   add_compile_options(/Zc:__cplusplus /MP3)
else()
   add_compile_definitions(LINUX_MAC NO_GCC_PRAGMA)
   add_compile_options(-fno-strict-aliasing)
endif()
if(GMAT_TUTORIAL_BUILD_BASE)
   find_package(XercesC REQUIRED)
   find_package(Threads REQUIRED)
   set(CSPICE_INCLUDE_DIR "${CSPICE_DIR}/include")
   find_library(CSPICE_LIB NAMES cspice cspice.a PATHS "${CSPICE_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
   set(CSPICE_LIB_DEBUG "${CSPICE_LIB}")
   if(APPLE)
      find_library(COREFOUNDATION_LIBRARY CoreFoundation REQUIRED)
      find_library(CORESERVICES_LIBRARY CoreServices REQUIRED)
   endif()
   function(_ADDSOURCEGROUPS)
   endfunction()
   function(_SETOUTPUTDIRECTORY target directory)
      set_target_properties(${target} PROPERTIES
         RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${directory}"
         LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${directory}")
   endfunction()
   if(APPLE)
      add_subdirectory("${GMAT_SOURCE_DIR}/src/gmatutil" util)
      set_target_properties(GmatUtil PROPERTIES VERSION R2026a)
   endif()
   add_subdirectory("${GMAT_SOURCE_DIR}/src/base" core)
   set_target_properties(GmatBase PROPERTIES VERSION R2026a)
endif()
set(GMAT_QT_TUTORIAL_ONLY ON)
add_subdirectory("${GMAT_SOURCE_DIR}/src/qtgui" frontend)

option(GMAT_EVENT_RESULT_TESTS "Build the independent eclipse and contact checks" OFF)
if(GMAT_EVENT_RESULT_TESTS)
   find_package(Qt6 REQUIRED COMPONENTS Test)
   find_library(event_cspice NAMES cspice cspice.a PATHS "${CSPICE_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
   foreach(family Eclipse Contact)
      # The current editor uses the window close action for dirty-close prompts.
      # Retain the protected test and replace only its obsolete hidden-button gesture.
      file(READ "${GMAT_SOURCE_DIR}/src/qtgui/tests/Nasa${family}ResultTests.cpp" event_source)
      string(REPLACE "click(buttons,QDialogButtonBox::Close)" "child->close(); events()"
         event_source "${event_source}")
      file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/Nasa${family}ResultCurrent.cpp" "${event_source}")
      add_executable(GmatQtNasa${family}ResultTests
         "${CMAKE_CURRENT_BINARY_DIR}/Nasa${family}ResultCurrent.cpp")
      target_include_directories(GmatQtNasa${family}ResultTests PRIVATE "${GMAT_SOURCE_DIR}/src/qtgui/tests")
      target_link_libraries(GmatQtNasa${family}ResultTests PRIVATE GmatQtFrontend Qt6::Test "${event_cspice}")
   endforeach()
endif()

option(GMAT_TUTORIAL_PATCH_SMOOTHER "Build the targeted smoother repair against archive libraries" OFF)
if(GMAT_TUTORIAL_PATCH_SMOOTHER)
   if(APPLE)
      set(gmat_plugin_dir "${GMAT_INSTALLATION}/Contents/Frameworks")
   else()
      set(gmat_plugin_dir "${GMAT_INSTALLATION}/plugins")
   endif()
   if(WIN32)
      find_library(gmat_estimation NAMES GmatEstimation PATHS "${GMAT_IMPORT_LIBRARIES}" NO_DEFAULT_PATH REQUIRED)
   elseif(APPLE)
      find_library(gmat_estimation NAMES GmatEstimation.R2026a PATHS "${gmat_plugin_dir}" NO_DEFAULT_PATH REQUIRED)
   else()
      find_library(gmat_estimation NAMES GmatEstimation PATHS "${gmat_plugin_dir}" NO_DEFAULT_PATH REQUIRED)
   endif()
   add_library(GmatEstimation UNKNOWN IMPORTED)
   set_target_properties(GmatEstimation PROPERTIES IMPORTED_LOCATION "${gmat_estimation}")
   file(GLOB estimation_directories LIST_DIRECTORIES true "${GMAT_SOURCE_DIR}/plugins/EstimationPlugin/src/base/*")
   set(ekf_directory "${GMAT_SOURCE_DIR}/plugins/ExtendedKalmanFilterPlugin/src/base")
   file(GLOB ekf_sources "${ekf_directory}/*/*.cpp")
   file(GLOB ekf_directories LIST_DIRECTORIES true "${ekf_directory}/*")
   add_library(TutorialSmootherPatch SHARED ${ekf_sources})
   target_include_directories(TutorialSmootherPatch PRIVATE ${gmat_header_directories})
   foreach(directory IN LISTS estimation_directories ekf_directories)
      if(IS_DIRECTORY "${directory}")
         target_include_directories(TutorialSmootherPatch PRIVATE "${directory}")
      endif()
   endforeach()
   target_compile_definitions(TutorialSmootherPatch PRIVATE KALMAN_EXPORTS _DYNAMICLINK NOMINMAX)
   target_link_libraries(TutorialSmootherPatch PRIVATE GmatBase GmatUtil GmatEstimation)
   set_target_properties(TutorialSmootherPatch PROPERTIES OUTPUT_NAME EKF PREFIX "lib" VERSION R2026a)
endif()
